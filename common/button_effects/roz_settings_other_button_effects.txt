#实用功能：外交权重平衡修正（添加）——常驻按钮，与“移除”按钮同时显示，非互斥
roz_settings_diplo_weight_add = {
    potential = { always = yes }
    allow = { always = yes }
    effect = {
        custom_tooltip = roz_start.7777.diplo_weight_add.ct
        hidden_effect = {
            from? = {
                add_modifier = { modifier = roz_diplo_weight_debuff }
            }
        }
    }
}

#实用功能：L星门封锁（布尔按钮·封锁态；与启用态同坐标互斥）
#  下方 every_system / every_system_megastructure 的 limit = { … } 是引擎过滤器关键字（非 scripted 定义），用于仅命中 L 星团星门。
roz_settings_lgate_disable = {
    potential = {
        from? = {
            is_ai = no
            has_country_flag = roz_lgate_spawn
            has_country_flag = roz_lgate_enable
            NOT = { has_country_flag = roz_lgate_disable }
        }
    }
    allow = { always = yes }
    effect = {
        custom_tooltip = roz_start.7777.lgate_disable.ct
        hidden_effect = {
            from? = {
                set_country_flag = roz_lgate_disable
                remove_country_flag = roz_lgate_enable
                every_system = {
                    if = {
                        limit = {
                            has_star_flag = lcluster1
                            has_star_flag = lcluster_lgate
                        }
                        every_system_megastructure = {
                            if = {
                                limit = {
                                    is_megastructure_type = lgate_base
                                }
                                remove_megastructure = this
                            }
                        }
                        spawn_megastructure = {
                            type = lgate_base
                            orbit_angle = 225
                            orbit_distance = 360
                        }
                    }
                }
            }
        }
    }
}
#实用功能：L星门启用（布尔按钮·启用态）
roz_settings_lgate_enable = {

        from? = {
            is_ai = no
            has_country_flag = roz_lgate_spawn
            has_country_flag = roz_lgate_disable
            NOT = { has_country_flag = roz_lgate_enable }
        }
    }
    allow = { always = yes }
    effect = {
        custom_tooltip = roz_start.7777.lgate_enable.ct
        hidden_effect = {
            from? = {
                set_country_flag = roz_lgate_enable
                remove_country_flag = roz_lgate_disable
                every_system = {
                    if = {
                        limit = {
                            has_star_flag = lcluster1
                            has_star_flag = lcluster_lgate
                        }
                        every_system_megastructure = {
                            if = {
                                limit = {
                                    is_megastructure_type = lgate_base
                                    NOT = { has_megastructure_flag = lgate_activated }
                                }
                                activate_gateway = this
                                set_megastructure_flag = lgate_activated
                            }
                        }
                    }
                }
            }
        }
    }
}

#实用功能：国民性添加（原法令 ge_civic_edicts）
#  强制移除再写回国策的顺序与次数与原法令完全一致，请勿“优化”成单次调用。
roz_settings_civic_restore = {
    potential = { always = yes }
    allow = {
        from? = {
            custom_tooltip = {
                fail_text = roz_start.7777.civic_restore.fail
                has_country_flag = roz_leader_suigintou_get
                has_ascension_perk = ap_ge_2
                has_technology = GE_tech_reactor
                has_country_flag = ge_ap2_complate
                NOR = {
                    has_civic = civic_cor_celestial_military
                    has_civic = civic_cor_time_honored
                }
            }
        }
    }
    effect = {
        custom_tooltip = roz_start.7777.civic_restore.ct
        hidden_effect = {
            from? = {
                force_remove_civic = civic_cor_celestial_military
                force_remove_civic = civic_cor_time_honored
                force_remove_civic = civic_cor_celestial_military

                force_add_civic = civic_cor_celestial_military
                force_add_civic = civic_cor_time_honored
                force_add_civic = civic_cor_celestial_military
            }
        }
    }
}

#维护约定：
#  1. roz_settings_diplo_weight_add 与 roz_settings_diplo_weight_remove 是“常驻并列”按钮：
#     potential 恒为 always = yes，两者始终同时显示、同时可点击；效果为幂等操作（重复点击不会叠加修正）。
#  2. 只有 roz_settings_lgate_disable / roz_settings_lgate_enable 构成同坐标互斥布尔按钮（靠 potential 切换显隐）。
#  3. roz_settings_mutiplayer_tech 与 roz_settings_site_star 的 potential 恒为 always = yes（永远显示），
#     原法令的多人/剧情门槛写在 allow 内：条件不满足时按钮显示为灰色不可点，条件满足后才能执行效果。
#  4. 第二批整合按钮（roz_settings_ge_site / roz_settings_restart_boost_plan / roz_settings_ogas_contact /
#     roz_settings_civic_restore）遵循同一约定：potential 恒为 always = yes 保证常驻显示，
#     原法令 potential 的条件原样搬进 allow，并包在 custom_tooltip = { fail_text = … } 里输出本地化提示。
#  5. 效果块内的语句顺序是对原法令 effect 的逐行搬迁，尤其 roz_settings_civic_restore 的
#     三次 force_remove_civic / 三次 force_add_civic 不要合并或去重，否则国策重排行为可能与法令不一致。
